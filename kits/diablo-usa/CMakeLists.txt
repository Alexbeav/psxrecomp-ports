cmake_minimum_required(VERSION 3.20)
project(diablo_usa C CXX)

set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(PSXRECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/psxrecomp-src" CACHE PATH "PSXRecomp source")
set(RECOMP_UI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/recomp-ui" CACHE PATH "recomp-ui source")
set(PSX_RECOMP_UI ON CACHE BOOL "" FORCE)
set(PSX_REWIND ON CACHE BOOL "" FORCE)
set(PSX_NETPLAY OFF CACHE BOOL "" FORCE)
set(PSXRECOMP_BIOS_STEMS "SCPH1001" CACHE STRING "" FORCE)
set(PSXRECOMP_BIOS_STEM "SCPH1001" CACHE STRING "" FORCE)
set(PSXRECOMP_BIOS_PROFILE "${PSXRECOMP_ROOT}/bios/SCPH1001.toml" CACHE FILEPATH "" FORCE)
include("${PSXRECOMP_ROOT}/runtime/runtime.cmake")

file(GLOB GAME_FULL CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/generated/SLUS_006.19_full.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/generated/SLUS_006.19_full_*.c")
set(GAME_DISPATCH "${CMAKE_CURRENT_SOURCE_DIR}/generated/SLUS_006.19_dispatch.c")

psxrecomp_add_runtime_target(psx-runtime
    GAME_GENERATED_FULL_C ${GAME_FULL}
    GAME_GENERATED_DISPATCH_C "${GAME_DISPATCH}"
    WINDOW_TITLE "Diablo Recompiled"
    EXE_NAME "Diablo"
    GAME_VERSION "0.3.4"
    DEFAULT_GAME_CONFIG_PATH "game.toml")
